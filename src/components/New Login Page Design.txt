import React, { useState, useEffect } from 'react';
import { User as UserType } from '../types';
import { login, register } from '../utils/auth';
import { Eye, EyeOff, X, User, Lock, Mail, AlertCircle, Clock, Sparkles, Shield, Key } from 'lucide-react';

// EXPIRY DATE CONFIGURATION - Change this to extend or restrict access
// Set Year - Month - Day Than Set Time 23:59:59
const EXPIRY_DATE = new Date('2025-11-31T23:59:59');
//  Access Model Modify time in day 12:00:00

interface LoginProps {
  onClose: () => void;
  onLogin: (user: UserType) => void;
}

const Login: React.FC<LoginProps> = ({ onClose, onLogin }) => {
  const [isRegister, setIsRegister] = useState(false);
  const [formData, setFormData] = useState({
    email: '',
    password: '',
    name: ''
  });
  const [showPassword, setShowPassword] = useState(false);
  const [error, setError] = useState('');
  const [loading, setLoading] = useState(false);
  const [debugInfo, setDebugInfo] = useState('');
  const [isExpired, setIsExpired] = useState(false);
  const [showContactModal, setShowContactModal] = useState(false);

  // Check if current date is past expiry date
  useEffect(() => {
    const checkExpiry = () => {
      const now = new Date();
      setIsExpired(now > EXPIRY_DATE);
    };

    checkExpiry();
    // Check every minute to handle cases where user keeps the page open
    const interval = setInterval(checkExpiry, 60000);
    
    return () => clearInterval(interval);
  }, []);

  const handleSubmit = async (e: React.FormEvent) => {
    e.preventDefault();
    
    // Prevent submission if expired
    if (isExpired) {
      setError('Your Login API is expires Update API');
      return;
    }

    setLoading(true);
    setError('');
    setDebugInfo('');

    try {
      let user: UserType | null = null;

      if (isRegister) {
        if (!formData.name.trim()) {
          setError('Name is required');
          setLoading(false);
          return;
        }
        user = await register({
          email: formData.email,
          password: formData.password,
          name: formData.name,
          role: 'customer' // Default role for registration
        });
        if (!user) {
          setError('User already exists with this email');
          setLoading(false);
          return;
        }
      } else {
        // Add debug info for login
        setDebugInfo(`Attempting login with email: ${formData.email}`);
        
        user = await login(formData.email, formData.password);
        
        if (!user) {
          setError('Invalid email or password');
          setDebugInfo('Login failed - user is null');
          setLoading(false);
          return;
        }
        
        setDebugInfo(`Login successful! User role: ${user.role}, ID: ${user.id}`);
      }

      // Add debug info before calling onLogin
      setDebugInfo(`About to call onLogin with user: ${JSON.stringify(user, null, 2)}`);
      
      onLogin(user);
      onClose();
    } catch (err) {
      console.error('Login/Register error:', err);
      setError('An error occurred. Please try again.');
      setDebugInfo(`Error details: ${err instanceof Error ? err.message : 'Unknown error'}`);
    } finally {
      setLoading(false);
    }
  };

  const handleInputChange = (e: React.ChangeEvent<HTMLInputElement>) => {
    setFormData(prev => ({
      ...prev,
      [e.target.name]: e.target.value
    }));
    setError('');
    setDebugInfo('');
  };

  return (
    <div className="fixed inset-0 bg-gradient-to-br from-slate-900/95 via-purple-900/90 to-slate-900/95 backdrop-blur-xl flex items-center justify-center z-50 p-4">
      {/* Animated background particles */}
      <div className="absolute inset-0 overflow-hidden">
        <div className="absolute top-1/4 left-1/4 w-2 h-2 bg-cyan-400 rounded-full animate-ping opacity-30" style={{animationDelay: '0s'}}></div>
        <div className="absolute top-1/3 right-1/4 w-1 h-1 bg-purple-400 rounded-full animate-ping opacity-40" style={{animationDelay: '1s'}}></div>
        <div className="absolute bottom-1/4 left-1/3 w-1.5 h-1.5 bg-blue-400 rounded-full animate-ping opacity-30" style={{animationDelay: '2s'}}></div>
        <div className="absolute top-1/2 right-1/3 w-1 h-1 bg-pink-400 rounded-full animate-ping opacity-40" style={{animationDelay: '0.5s'}}></div>
      </div>

      {/* Main Modal Container */}
      <div className="relative w-full max-w-md animate-in zoom-in-95 duration-500">
        {/* Glowing border effect */}
        <div className="absolute inset-0 bg-gradient-to-r from-cyan-400 via-purple-500 to-pink-500 rounded-3xl blur-xl opacity-30 animate-pulse"></div>
        
        <div className="relative bg-gradient-to-br from-slate-900/95 via-slate-800/95 to-slate-900/95 backdrop-blur-2xl border border-slate-700/50 rounded-3xl shadow-2xl shadow-black/50 overflow-hidden">
          {/* Header Section */}
          <div className="relative p-8 text-center border-b border-slate-700/50">
            {/* Background pattern */}
            <div className="absolute inset-0 opacity-10">
              <div className="absolute inset-0 bg-[url('data:image/svg+xml;base64,PHN2ZyB3aWR0aD0iNjAiIGhlaWdodD0iNjAiIHZpZXdCb3g9IjAgMCA2MCA2MCIgeG1sbnM9Imh0dHA6Ly93d3cudzMub3JnLzIwMDAvc3ZnIj48ZyBmaWxsPSJub25lIiBmaWxsLXJ1bGU9ImV2ZW5vZGQiPjxnIGZpbGw9IiNjY2MiIGZpbGwtb3BhY2l0eT0iMC4xIj48Y2lyY2xlIGN4PSIzMCIgY3k9IjMwIiByPSIyIi8+PC9nPjwvZz48L3N2Zz4=')] bg-repeat"></div>
            </div>
            
            {/* Close button */}
            <button
              onClick={onClose}
              className="absolute top-4 right-4 p-2 text-slate-400 hover:text-white transition-all duration-300 hover:bg-slate-700/50 rounded-xl group"
            >
              <X className="w-5 h-5 group-hover:scale-110 transition-transform duration-300" />
            </button>

            {/* Logo/Icon */}
            <div className="relative inline-flex items-center justify-center w-20 h-20 bg-gradient-to-br from-cyan-400 via-purple-500 to-pink-500 rounded-2xl mb-6 shadow-2xl shadow-cyan-500/25 animate-bounce" style={{animationDuration: '3s'}}>
              <Shield className="w-10 h-10 text-white" />
              <div className="absolute inset-0 rounded-2xl bg-gradient-to-r from-cyan-400 to-pink-500 opacity-30 blur-sm animate-ping"></div>
            </div>

            {/* Title */}
            <h2 className="text-3xl font-bold bg-gradient-to-r from-cyan-400 via-purple-400 to-pink-400 bg-clip-text text-transparent mb-2">
              {isRegister ? 'Create Account' : 'Welcome Back'}
            </h2>
            <p className="text-slate-400 text-sm">
              {isRegister ? 'Join our community today' : 'Sign in to your account'}
            </p>
          </div>

          {/* API Expired Alert */}
          {isExpired && (
            <div className="mx-6 mt-6 relative overflow-hidden animate-in slide-in-from-top-2 duration-500">
              {/* Premium Background with multiple gradient layers */}
              <div className="absolute inset-0 bg-gradient-to-br from-red-500/20 via-red-600/25 to-red-700/30 rounded-xl animate-pulse"></div>
              <div className="absolute inset-0 bg-gradient-to-tl from-orange-500/15 via-red-500/20 to-pink-500/15 rounded-xl animate-pulse" style={{animationDelay: '0.5s'}}></div>
              <div className="absolute inset-0 bg-gradient-to-tr from-yellow-500/10 via-orange-500/15 to-red-500/10 rounded-xl animate-pulse" style={{animationDelay: '1s'}}></div>
              <div className="absolute inset-0 opacity-30">
                <div className="w-full h-full bg-[url('data:image/svg+xml;base64,PHN2ZyB3aWR0aD0iNjAiIGhlaWdodD0iNjAiIHZpZXdCb3g9IjAgMCA2MCA2MCIgeG1sbnM9Imh0dHA6Ly93d3cudzMub3JnLzIwMDAvc3ZnIj48ZyBmaWxsPSJub25lIiBmaWxsLXJ1bGU9ImV2ZW5vZGQiPjxnIGZpbGw9IiNlZjQ0NDQiIGZpbGwtb3BhY2l0eT0iMC4xIj48Y2lyY2xlIGN4PSIzMCIgY3k9IjMwIiByPSIyIi8+PC9nPjwvZz48L3N2Zz4=')] bg-repeat animate-pulse" style={{animationDelay: '1.5s'}}></div>
              </div>
              
              {/* Enhanced glowing border effects */}
              <div className="absolute inset-0 rounded-xl bg-gradient-to-r from-red-400 via-red-500 to-red-600 opacity-25 blur-md animate-pulse"></div>
              <div className="absolute inset-0 rounded-xl bg-gradient-to-r from-orange-400 via-red-500 to-pink-500 opacity-15 blur-lg animate-pulse" style={{animationDelay: '0.7s'}}></div>
              
              {/* Floating particles effect */}
              <div className="absolute inset-0 overflow-hidden">
                <div className="absolute top-2 right-4 w-1 h-1 bg-red-300 rounded-full animate-bounce" style={{animationDelay: '0.3s', animationDuration: '2s'}}></div>
                <div className="absolute top-6 right-2 w-0.5 h-0.5 bg-orange-400 rounded-full animate-bounce" style={{animationDelay: '0.8s', animationDuration: '1.8s'}}></div>
                <div className="absolute top-4 left-3 w-0.5 h-0.5 bg-pink-400 rounded-full animate-bounce" style={{animationDelay: '1.2s', animationDuration: '2.2s'}}></div>
              </div>
              
              {/* Main alert content with premium styling */}
              <div className="relative p-6 border border-red-400/50 rounded-2xl backdrop-blur-xl bg-black/30 animate-in zoom-in-95 duration-700 delay-200 shadow-2xl shadow-red-500/20">
                {/* Premium Header with enhanced icon and title */}
                <div className="text-center mb-5">
                  <div className="relative inline-flex items-center justify-center w-16 h-16 bg-gradient-to-br from-red-400 via-red-500 to-red-600 rounded-2xl mb-4 shadow-2xl shadow-red-500/50 animate-bounce" style={{animationDelay: '0.5s', animationDuration: '2s'}}>
                    <Clock className="w-8 h-8 text-white animate-spin" style={{animationDuration: '3s'}} />
                    {/* Multiple glowing rings around icon */}
                    <div className="absolute inset-0 rounded-2xl bg-gradient-to-r from-red-400 to-red-600 opacity-40 blur-sm animate-ping"></div>
                    <div className="absolute inset-0 rounded-2xl bg-gradient-to-r from-orange-400 to-red-500 opacity-20 blur-md animate-pulse" style={{animationDelay: '0.5s'}}></div>
                  </div>
                  <h3 className="text-2xl font-bold bg-gradient-to-r from-red-200 via-red-300 to-red-400 bg-clip-text text-transparent mb-3 animate-in slide-in-from-bottom-2 duration-500 delay-300">
                    ⚠️ API Expired
                  </h3>
                  <div className="w-20 h-1 bg-gradient-to-r from-transparent via-red-400 via-red-500 to-transparent mx-auto rounded-full animate-in slide-in-from-left-2 duration-700 delay-500 shadow-lg shadow-red-500/30"></div>
                </div>
                
                {/* Premium Message Cards */}
                <div className="text-center mb-6 animate-in fade-in duration-700 delay-400 space-y-3">
                  <div className="bg-gradient-to-r from-red-500/25 to-red-600/25 border border-red-400/40 rounded-xl p-4 shadow-lg shadow-red-500/20 backdrop-blur-sm">
                    <p className="text-red-100 text-sm leading-relaxed font-bold animate-pulse">
                      🚫 Your Login API is expires Update API
                    </p>
                  </div>
                  <div className="bg-gradient-to-r from-orange-500/25 to-red-500/25 border border-orange-400/40 rounded-xl p-3 shadow-lg shadow-orange-500/20 backdrop-blur-sm">
                    <p className="text-orange-100 text-xs font-semibold animate-pulse" style={{animationDelay: '0.5s'}}>
                      📞 Contact administrator for renewal
                    </p>
                  </div>
                </div>
                
                {/* Premium Action buttons with enhanced effects */}
                <div className="flex gap-4 justify-center animate-in slide-in-from-bottom-2 duration-500 delay-600">
                  <button
                    onClick={onClose}
                    className="px-6 py-3 bg-gradient-to-r from-red-500 via-red-600 to-red-700 hover:from-red-600 hover:via-red-700 hover:to-red-800 text-white rounded-xl font-bold transition-all duration-300 hover:scale-110 hover:shadow-2xl shadow-xl shadow-red-500/40 border border-red-400/50 hover:border-red-300/70 text-sm transform hover:-translate-y-1 active:scale-95 relative overflow-hidden group"
                  >
                    <span className="relative z-10 flex items-center space-x-2">
                      <span>🔒</span>
                      <span>Close</span>
                    </span>
                    <div className="absolute inset-0 bg-gradient-to-r from-red-400 to-red-600 opacity-0 group-hover:opacity-100 transition-opacity duration-300 rounded-xl"></div>
                    <div className="absolute inset-0 bg-gradient-to-r from-red-300 to-red-500 opacity-0 group-hover:opacity-20 transition-opacity duration-300 rounded-xl blur-sm"></div>
                  </button>
                  <button
                    onClick={() => setShowContactModal(true)}
                    className="px-6 py-3 bg-gradient-to-r from-slate-600 via-slate-700 to-slate-800 hover:from-slate-700 hover:via-slate-800 hover:to-slate-900 text-white rounded-xl font-bold transition-all duration-300 hover:scale-110 hover:shadow-2xl shadow-xl shadow-slate-600/40 border border-slate-500/50 hover:border-slate-400/70 text-sm transform hover:-translate-y-1 active:scale-95 relative overflow-hidden group"
                  >
                    <span className="relative z-10 flex items-center space-x-2">
                      <span>📞</span>
                      <span>Contact</span>
                    </span>
                    <div className="absolute inset-0 bg-gradient-to-r from-slate-500 to-slate-700 opacity-0 group-hover:opacity-100 transition-opacity duration-300 rounded-xl"></div>
                    <div className="absolute inset-0 bg-gradient-to-r from-slate-400 to-slate-600 opacity-0 group-hover:opacity-20 transition-opacity duration-300 rounded-xl blur-sm"></div>
                  </button>
                </div>
                
                {/* Enhanced decorative elements with premium styling */}
                <div className="absolute top-3 right-3 w-3 h-3 bg-red-400 rounded-full animate-ping shadow-lg shadow-red-400/50" style={{animationDelay: '0.5s'}}></div>
                <div className="absolute bottom-3 left-3 w-2.5 h-2.5 bg-red-500 rounded-full animate-ping shadow-lg shadow-red-500/50" style={{animationDelay: '1s'}}></div>
                <div className="absolute top-4 right-6 w-2 h-2 bg-red-300 rounded-full animate-pulse shadow-lg shadow-red-300/50" style={{animationDelay: '1.5s'}}></div>
                <div className="absolute bottom-4 left-6 w-2 h-2 bg-red-400 rounded-full animate-pulse shadow-lg shadow-red-400/50" style={{animationDelay: '2s'}}></div>
                <div className="absolute top-1/2 right-2 w-1.5 h-1.5 bg-orange-400 rounded-full animate-pulse shadow-lg shadow-orange-400/50" style={{animationDelay: '0.8s'}}></div>
                <div className="absolute top-1/2 left-2 w-1.5 h-1.5 bg-pink-400 rounded-full animate-pulse shadow-lg shadow-pink-400/50" style={{animationDelay: '1.2s'}}></div>
                <div className="absolute top-1/4 right-1 w-1 h-1 bg-yellow-400 rounded-full animate-pulse shadow-lg shadow-yellow-400/50" style={{animationDelay: '1.8s'}}></div>
                <div className="absolute bottom-1/4 left-1 w-1 h-1 bg-purple-400 rounded-full animate-pulse shadow-lg shadow-purple-400/50" style={{animationDelay: '2.2s'}}></div>
              </div>
            </div>
          )}

          {/* Form Section */}
          <div className="p-8">
            <form onSubmit={handleSubmit} className="space-y-6">
              {isRegister && (
                <div className="relative group">
                  <div className="absolute inset-y-0 left-0 pl-4 flex items-center pointer-events-none">
                    <User className="w-5 h-5 text-slate-400 group-focus-within:text-cyan-400 transition-colors duration-300" />
                  </div>
                  <input
                    type="text"
                    name="name"
                    placeholder="Full Name"
                    value={formData.name}
                    onChange={handleInputChange}
                    className="w-full pl-12 pr-4 py-4 bg-slate-800/50 border border-slate-600/50 rounded-xl text-white placeholder-slate-400 focus:outline-none focus:ring-2 focus:ring-cyan-400/50 focus:border-cyan-400/50 transition-all duration-300 backdrop-blur-sm disabled:opacity-50 disabled:cursor-not-allowed"
                    required
                    disabled={isExpired}
                    onClick={() => {
                      if (isExpired) {
                        setError('Your Login API is expires Update API');
                      }
                    }}
                  />
                  <div className="absolute inset-0 rounded-xl bg-gradient-to-r from-cyan-400/0 via-cyan-400/0 to-cyan-400/0 group-focus-within:from-cyan-400/10 group-focus-within:via-cyan-400/5 group-focus-within:to-cyan-400/10 transition-all duration-300 pointer-events-none"></div>
                </div>
              )}

              <div className="relative group">
                <div className="absolute inset-y-0 left-0 pl-4 flex items-center pointer-events-none">
                  <Mail className="w-5 h-5 text-slate-400 group-focus-within:text-cyan-400 transition-colors duration-300" />
                </div>
                <input
                  type="email"
                  name="email"
                  placeholder="Email Address"
                  value={formData.email}
                  onChange={handleInputChange}
                  className="w-full pl-12 pr-4 py-4 bg-slate-800/50 border border-slate-600/50 rounded-xl text-white placeholder-slate-400 focus:outline-none focus:ring-2 focus:ring-cyan-400/50 focus:border-cyan-400/50 transition-all duration-300 backdrop-blur-sm disabled:opacity-50 disabled:cursor-not-allowed"
                  required
                  disabled={isExpired}
                  onClick={() => {
                    if (isExpired) {
                      setError('Your Login API is expires Update API');
                    }
                  }}
                />
                <div className="absolute inset-0 rounded-xl bg-gradient-to-r from-cyan-400/0 via-cyan-400/0 to-cyan-400/0 group-focus-within:from-cyan-400/10 group-focus-within:via-cyan-400/5 group-focus-within:to-cyan-400/10 transition-all duration-300 pointer-events-none"></div>
              </div>

              <div className="relative group">
                <div className="absolute inset-y-0 left-0 pl-4 flex items-center pointer-events-none">
                  <Lock className="w-5 h-5 text-slate-400 group-focus-within:text-cyan-400 transition-colors duration-300" />
                </div>
                <input
                  type={showPassword ? 'text' : 'password'}
                  name="password"
                  placeholder="Password"
                  value={formData.password}
                  onChange={handleInputChange}
                  className="w-full pl-12 pr-12 py-4 bg-slate-800/50 border border-slate-600/50 rounded-xl text-white placeholder-slate-400 focus:outline-none focus:ring-2 focus:ring-cyan-400/50 focus:border-cyan-400/50 transition-all duration-300 backdrop-blur-sm disabled:opacity-50 disabled:cursor-not-allowed"
                  required
                  disabled={isExpired}
                  onClick={() => {
                    if (isExpired) {
                      setError('Your Login API is expires Update API');
                    }
                  }}
                />
                <button
                  type="button"
                  onClick={() => setShowPassword(!showPassword)}
                  className="absolute right-3 top-1/2 transform -translate-y-1/2 p-2 text-slate-400 hover:text-white transition-all duration-300 hover:bg-slate-700/50 rounded-lg disabled:opacity-50 disabled:cursor-not-allowed"
                  disabled={isExpired}
                >
                  {showPassword ? <EyeOff className="w-5 h-5" /> : <Eye className="w-5 h-5" />}
                </button>
                <div className="absolute inset-0 rounded-xl bg-gradient-to-r from-cyan-400/0 via-cyan-400/0 to-cyan-400/0 group-focus-within:from-cyan-400/10 group-focus-within:via-cyan-400/5 group-focus-within:to-cyan-400/10 transition-all duration-300 pointer-events-none"></div>
              </div>

              {/* Error Display */}
              {error && (
                <div className="flex items-center space-x-3 p-4 bg-gradient-to-r from-red-500/20 to-red-600/20 border border-red-500/30 rounded-xl backdrop-blur-sm animate-in slide-in-from-top-2 duration-300">
                  <AlertCircle className="w-5 h-5 text-red-400 flex-shrink-0" />
                  <span className="text-red-300 text-sm font-medium">{error}</span>
                </div>
              )}

              {/* Debug Info */}
              {debugInfo && (
                <div className="p-4 bg-gradient-to-r from-cyan-500/20 to-blue-500/20 border border-cyan-500/30 rounded-xl backdrop-blur-sm">
                  <p className="text-cyan-300 text-xs font-mono">{debugInfo}</p>
                </div>
              )}

              {/* Submit Button */}
              <button
                type="submit"
                disabled={loading || isExpired}
                className="w-full px-6 py-4 bg-gradient-to-r from-cyan-500 via-purple-500 to-pink-500 hover:from-cyan-600 hover:via-purple-600 hover:to-pink-600 text-white rounded-xl font-bold transition-all duration-300 hover:scale-105 hover:shadow-2xl shadow-xl shadow-cyan-500/25 disabled:opacity-50 disabled:cursor-not-allowed disabled:scale-100 relative overflow-hidden group"
              >
                <span className="relative z-10 flex items-center justify-center space-x-2">
                  {loading ? (
                    <>
                      <div className="w-5 h-5 border-2 border-white/30 border-t-white rounded-full animate-spin"></div>
                      <span>Processing...</span>
                    </>
                  ) : isExpired ? (
                    <>
                      <span>🔒</span>
                      <span>Login Disabled</span>
                    </>
                  ) : (
                    <>
                      <Key className="w-5 h-5" />
                      <span>{isRegister ? 'Create Account' : 'Sign In'}</span>
                    </>
                  )}
                </span>
                <div className="absolute inset-0 bg-gradient-to-r from-cyan-400 to-pink-500 opacity-0 group-hover:opacity-100 transition-opacity duration-300 rounded-xl"></div>
              </button>
            </form>

            {/* Toggle Section */}
            <div className="mt-8 text-center">
              <button
                onClick={() => setIsRegister(!isRegister)}
                className="text-cyan-400 hover:text-cyan-300 text-sm transition-all duration-300 hover:scale-105 disabled:opacity-50 disabled:cursor-not-allowed disabled:scale-100"
                disabled={isExpired}
              >
                {isRegister 
                  ? 'Already have an account? Sign in'
                  : 'Need an account? Create one'
                }
              </button>
            </div>
          </div>
        </div>
      </div>

      {/* Beautiful Contact Modal */}
      {showContactModal && (
        <div className="fixed inset-0 bg-black/50 backdrop-blur-sm flex items-center justify-center z-[60] p-4 animate-in fade-in duration-300">
          <div className="bg-gradient-to-br from-slate-900 via-slate-800 to-slate-900 border border-cyan-400/30 rounded-2xl max-w-md w-full shadow-2xl animate-in zoom-in-95 duration-300">
            {/* Header */}
            <div className="relative p-6 border-b border-cyan-400/20">
              <div className="text-center">
                <div className="inline-flex items-center justify-center w-16 h-16 bg-gradient-to-br from-cyan-400 to-blue-500 rounded-full mb-4 shadow-lg shadow-cyan-500/25 animate-bounce">
                  <svg className="w-8 h-8 text-white" fill="none" stroke="currentColor" viewBox="0 0 24 24">
                    <path strokeLinecap="round" strokeLinejoin="round" strokeWidth={2} d="M16 7a4 4 0 11-8 0 4 4 0 018 0zM12 14a7 7 0 00-7 7h14a7 7 0 00-7-7z" />
                  </svg>
                </div>
                <h3 className="text-2xl font-bold bg-gradient-to-r from-cyan-400 to-blue-500 bg-clip-text text-transparent mb-2">
                  Developer Contact
                </h3>
                <div className="w-20 h-0.5 bg-gradient-to-r from-transparent via-cyan-400 to-transparent mx-auto"></div>
              </div>
              
              {/* Close button */}
              <button
                onClick={() => setShowContactModal(false)}
                className="absolute top-4 right-4 p-2 text-slate-400 hover:text-white transition-colors rounded-lg hover:bg-slate-700/50"
              >
                <X className="w-5 h-5" />
              </button>
            </div>

            {/* Content */}
            <div className="p-6">
              {/* Developer Info */}
              <div className="space-y-4 mb-6">
                <div className="bg-gradient-to-r from-cyan-500/10 to-blue-500/10 border border-cyan-400/20 rounded-xl p-4">
                  <div className="flex items-center space-x-3">
